function getCurrentLocation() { let status = document.getElementById("location-status"); if (!navigator.geolocation) { status.innerText = "Your browser does not support location."; return; } status.innerText = "Getting your location..."; navigator.geolocation.getCurrentPosition( function(position) { let latitude = position.coords.latitude; let longitude = position.coords.longitude; let deliveryMapLink = "https://www.google.com/maps/dir/?api=1&destination=" + latitude + "%2C" + longitude; document.getElementById("customer-location").value = deliveryMapLink; status.innerText = "Current location added successfully"; document.getElementById("map").innerHTML = ""; }, function() { status.innerText = "Location permission denied. Please type your address."; }, { enableHighAccuracy: true, timeout: 15000, maximumAge: 0 } ); }
